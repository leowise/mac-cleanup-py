# TUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Textual-based interactive TUI to mac-cleanup-py with three screens: module selection, live progress dashboard, and results summary.

**Architecture:** A new `mac_cleanup/tui/` package is the entire presentation layer. The existing cleanup engine (`core.py`, `config.py`, `main.py`) is untouched except for a `CleanupRunner` bridge in `tui/runner.py` that wraps the sync engine with async callbacks. Entry point launches TUI when `mac-cleanup` is run bare or with the `ui` subcommand.

**Tech Stack:** `textual>=0.60`, Python 3.11+, existing `rich`, `attrs`

## Global Constraints

- Python `>=3.10,<3.15`
- `textual>=0.60` added to `[tool.poetry.dependencies]` in `pyproject.toml`
- Line length: 120 chars (matches `[tool.black]` config)
- All new source files must pass `pyright` strict mode (match existing `pyproject.toml` settings)
- Existing CLI subcommands (`cleanup`, `dry-run`, `restore`, `configure`, `custom-path`, `update`) must continue working unchanged
- No modifications to `core.py`, `core_modules.py`, `default_modules.py`, `progress.py`, `backup.py`

```python
"""Async bridge between TUI and the sync cleanup engine."""

import asyncio
from dataclasses import dataclass, field
from os import statvfs
from pathlib import Path
from typing import Callable, Optional

from mac_cleanup.config import Config
from mac_cleanup.core import _Collector


@dataclass
class ModuleResult:
    name: str
    status: str  # "done" | "error" | "skipped"
    error: Optional[str] = None


@dataclass
class CleanupResult:
    module_results: list[ModuleResult] = field(default_factory=list)
    total_bytes_freed: float = 0.0
    dry_run: bool = False


OnStartCallback = Callable[[str], None]
OnCompleteCallback = Callable[[ModuleResult], None]


class CleanupRunner:
    """Wraps the sync cleanup engine for use from an async Textual app."""

    def __init__(self, config_path: Path) -> None:
        self._config_path = config_path

    def _run_sync(
        self,
        enabled_modules: list[str],
        dry_run: bool,
        on_start: OnStartCallback,
        on_complete: OnCompleteCallback,
    ) -> CleanupResult:
        # Reset Borg shared state so each TUI run starts clean
        _Collector._shared_instance.clear()

        config = Config(config_path_=self._config_path)
        all_modules = config.get_modules

        result = CleanupResult(dry_run=dry_run)
        free_before = float(statvfs("/").f_bavail * statvfs("/").f_frsize)

        for module_name in enabled_modules:
            module_fn = all_modules.get(module_name)
            if module_fn is None:
                mod_result = ModuleResult(name=module_name, status="skipped")
                result.module_results.append(mod_result)
                on_complete(mod_result)
                continue

            on_start(module_name)

            try:
                # Reset collector so this module runs in isolation
                _Collector._shared_instance.clear()
                module_fn()

                if not dry_run:
                    collector = _Collector()
                    for unit in collector._execute_list:  # noqa: SLF001
                        for mod in unit.modules:
                            try:
                                mod._execute()  # noqa: SLF001
                            except Exception:
                                pass

                mod_result = ModuleResult(name=module_name, status="done")
            except Exception as exc:
                mod_result = ModuleResult(name=module_name, status="error", error=str(exc))

            result.module_results.append(mod_result)
            on_complete(mod_result)

        free_after = float(statvfs("/").f_bavail * statvfs("/").f_frsize)
        result.total_bytes_freed = max(0.0, free_after - free_before)
        return result

    async def run(
        self,
        enabled_modules: list[str],
        dry_run: bool,
        on_start: OnStartCallback,
        on_complete: OnCompleteCallback,
    ) -> CleanupResult:
        """Run cleanup in a thread, reporting progress via callbacks."""
        return await asyncio.to_thread(
            self._run_sync, enabled_modules, dry_run, on_start, on_complete
        )
```

```python
"""Scrollable checklist widget for selecting cleanup modules."""

from textual.widgets import SelectionList
from textual.widgets.selection_list import Selection


class ModuleList(SelectionList[str]):
    """Scrollable checklist of cleanup module names."""

    DEFAULT_CSS = """
    ModuleList {
        height: 1fr;
        border: solid $accent;
    }
    """

    def build_selections(
        self,
        all_modules: list[str],
        enabled_modules: list[str],
    ) -> list[Selection[str]]:
        """Build sorted Selection items, pre-checking enabled modules."""
        enabled_set = set(enabled_modules)
        return [
            Selection(name, name, initial_state=(name in enabled_set))
            for name in sorted(all_modules)
        ]
```

```python
"""Module selection screen — first screen of the TUI."""

from pathlib import Path
from typing import Optional

from textual.app import ComposeResult
from textual.message import Message
from textual.screen import Screen
from textual.widgets import Footer, Header, Label

from mac_cleanup.tui.widgets.module_list import ModuleList


class RunRequested(Message):
    """Posted when user confirms module selection and wants to run cleanup."""

    def __init__(self, enabled_modules: list[str], dry_run: bool) -> None:
        super().__init__()
        self.enabled_modules = enabled_modules
        self.dry_run = dry_run


class SelectionScreen(Screen[None]):
    """Scrollable module checklist with keyboard controls."""

    BINDINGS = [
        ("space", "toggle_module", "Toggle"),
        ("a", "select_all", "Select all"),
        ("n", "deselect_all", "Deselect all"),
        ("d", "toggle_dry_run", "Toggle dry-run"),
        ("r,enter", "run", "Run"),
        ("q", "quit", "Quit"),
    ]

    def __init__(
        self,
        all_modules: list[str],
        enabled_modules: list[str],
        name: Optional[str] = None,
        id: Optional[str] = None,  # noqa: A002
        classes: Optional[str] = None,
    ) -> None:
        super().__init__(name=name, id=id, classes=classes)
        self._all_modules = all_modules
        self._enabled_modules = enabled_modules
        self.dry_run: bool = False

    def compose(self) -> ComposeResult:
        yield Header(show_clock=True)
        yield Label("mac-cleanup  |  [bold]space[/bold] toggle  [bold]a[/bold] all  [bold]n[/bold] none  "
                    "[bold]d[/bold] dry-run  [bold]r[/bold] run  [bold]q[/bold] quit", id="hint")
        self._module_list = ModuleList(
            *self._module_list_widget().build_selections(self._all_modules, self._enabled_modules)
        )
        yield self._module_list
        yield Footer()

    def _module_list_widget(self) -> ModuleList:
        return ModuleList()

    def on_mount(self) -> None:
        selections = ModuleList().build_selections(self._all_modules, self._enabled_modules)
        self.query_one(ModuleList).clear_options()
        for sel in selections:
            self.query_one(ModuleList).add_option(sel)
        self._update_header()

    def _update_header(self) -> None:
        mode = "[yellow]DRY-RUN[/yellow]" if self.dry_run else "[green]LIVE[/green]"
        self.sub_title = f"Mode: {mode}"

    def action_toggle_dry_run(self) -> None:
        self.dry_run = not self.dry_run
        self._update_header()

    def action_select_all(self) -> None:
        self.query_one(ModuleList).select_all()

    def action_deselect_all(self) -> None:
        self.query_one(ModuleList).deselect_all()

    def action_run(self) -> None:
        selected = list(self.query_one(ModuleList).selected)
        if not selected:
            self.notify("Select at least one module.", severity="warning")
            return
        self.post_message(RunRequested(enabled_modules=selected, dry_run=self.dry_run))

    def action_quit(self) -> None:
        self.app.exit()
```

```python
"""Live progress dashboard — second screen of the TUI."""

from pathlib import Path
from typing import Optional

from textual import work
from textual.app import ComposeResult
from textual.message import Message
from textual.screen import Screen
from textual.widgets import Footer, Header, Label, ProgressBar, RichLog

from mac_cleanup.tui.runner import CleanupResult, CleanupRunner, ModuleResult


class CleanupDone(Message):
    """Posted when the cleanup runner finishes."""

    def __init__(self, result: CleanupResult) -> None:
        super().__init__()
        self.result = result


class ProgressScreen(Screen[None]):
    """Displays live progress while cleanup runs in the background."""

    BINDINGS = [("q", "quit", "Quit")]

    def __init__(
        self,
        enabled_modules: list[str],
        dry_run: bool,
        config_path: Path,
        name: Optional[str] = None,
        id: Optional[str] = None,  # noqa: A002
        classes: Optional[str] = None,
    ) -> None:
        super().__init__(name=name, id=id, classes=classes)
        self._enabled_modules = enabled_modules
        self._dry_run = dry_run
        self._config_path = config_path
        self._total = len(enabled_modules)
        self._done = 0

    def compose(self) -> ComposeResult:
        mode = "DRY-RUN" if self._dry_run else "LIVE"
        yield Header(show_clock=True)
        yield Label(f"Running cleanup ({mode}) — {self._total} modules", id="status-label")
        yield ProgressBar(total=self._total, show_eta=True, id="overall-progress")
        yield RichLog(highlight=True, markup=True, id="log")
        yield Footer()

    def on_mount(self) -> None:
        self._run_cleanup()

    @work(exclusive=True)
    async def _run_cleanup(self) -> None:
        runner = CleanupRunner(config_path=self._config_path)
        log = self.query_one(RichLog)
        bar = self.query_one(ProgressBar)

        def on_start(module_name: str) -> None:
            self.call_from_thread(log.write, f"[cyan]⚙[/cyan]  {module_name}...")

        def on_complete(mod_result: ModuleResult) -> None:
            icon = "[green]✓[/green]" if mod_result.status == "done" else "[red]✗[/red]"
            msg = f"{icon}  {mod_result.name}"
            if mod_result.error:
                msg += f" — [red]{mod_result.error}[/red]"
            self.call_from_thread(log.write, msg)
            self.call_from_thread(bar.advance, 1)

        result = await runner.run(
            enabled_modules=self._enabled_modules,
            dry_run=self._dry_run,
            on_start=on_start,
            on_complete=on_complete,
        )
        self.post_message(CleanupDone(result=result))

    def action_quit(self) -> None:
        self.app.exit()
```

```python
"""Results summary screen — third screen of the TUI."""

from typing import Optional

from textual.app import ComposeResult
from textual.screen import Screen
from textual.widgets import DataTable, Footer, Header, Label

from mac_cleanup.tui.runner import CleanupResult
from mac_cleanup.utils import bytes_to_human


class ResultsScreen(Screen[None]):
    """Displays total space freed and per-module status after cleanup."""

    BINDINGS = [
        ("r", "run_again", "Run again"),
        ("b", "back", "Back to selection"),
        ("q", "quit", "Quit"),
    ]

    def __init__(
        self,
        result: CleanupResult,
        name: Optional[str] = None,
        id: Optional[str] = None,  # noqa: A002
        classes: Optional[str] = None,
    ) -> None:
        super().__init__(name=name, id=id, classes=classes)
        self._result = result

    def compose(self) -> ComposeResult:
        mode = "DRY-RUN" if self._result.dry_run else "LIVE"
        freed = bytes_to_human(self._result.total_bytes_freed)
        yield Header(show_clock=True)
        yield Label(
            f"[bold green]Cleanup complete ({mode})[/bold green]\n\n"
            f"Total freed: [bold cyan]{freed}[/bold cyan]",
            id="summary",
        )
        table = DataTable(id="module-table")
        yield table
        yield Footer()

    def on_mount(self) -> None:
        table = self.query_one(DataTable)
        table.add_columns("Module", "Status", "Error")
        for mod in self._result.module_results:
            status_markup = {
                "done": "[green]✓ done[/green]",
                "error": "[red]✗ error[/red]",
                "skipped": "[yellow]— skipped[/yellow]",
            }.get(mod.status, mod.status)
            table.add_row(mod.name, status_markup, mod.error or "")

    def action_run_again(self) -> None:
        from mac_cleanup.tui.screens.selection import RunRequested
        # Pop back to selection screen; app handles RunRequested there
        self.app.pop_screen()
        self.app.pop_screen()

    def action_back(self) -> None:
        self.app.pop_screen()
        self.app.pop_screen()

    def action_quit(self) -> None:
        self.app.exit()
```

```python
"""Main Textual application — owns screen transitions."""

from os import environ
from pathlib import Path
from typing import Optional

from textual.app import App, ComposeResult

from mac_cleanup.tui.screens.progress import CleanupDone, ProgressScreen
from mac_cleanup.tui.screens.results import ResultsScreen
from mac_cleanup.tui.screens.selection import RunRequested, SelectionScreen


def _default_config_path() -> Path:
    if (config_home := environ.get("XDG_CONFIG_HOME")) is not None:
        return Path(config_home).expanduser() / "mac_cleanup_py" / "config.toml"
    return Path.home() / ".mac_cleanup_py"


class MacCleanupApp(App[None]):
    """Interactive TUI for mac-cleanup-py."""

    TITLE = "mac-cleanup"
    SUB_TITLE = "macOS cleanup utility"
    CSS = """
    #hint { color: $text-muted; padding: 0 1; }
    #summary { padding: 1 2; }
    #status-label { padding: 0 1; color: $text-muted; }
    """

    def __init__(
        self,
        config_path: Optional[Path] = None,
        driver_class: Optional[type] = None,
        css_path: Optional[str] = None,
        watch_css: bool = False,
    ) -> None:
        super().__init__(driver_class=driver_class, css_path=css_path, watch_css=watch_css)
        self._config_path = config_path or _default_config_path()
        self._all_modules: list[str] = []
        self._enabled_modules: list[str] = []

    def on_mount(self) -> None:
        self._load_modules()
        self.push_screen(
            SelectionScreen(
                all_modules=self._all_modules,
                enabled_modules=self._enabled_modules,
            )
        )

    def _load_modules(self) -> None:
        """Load module names from config (best-effort; shows all default modules on failure)."""
        try:
            from mac_cleanup.config import Config
            from mac_cleanup.core import _Collector
            _Collector._shared_instance.clear()
            config = Config(config_path_=self._config_path)
            self._all_modules = sorted(config.get_modules.keys())
            self._enabled_modules = list(config.get_config_data.get("enabled", []))
        except Exception:
            from mac_cleanup import default_modules
            from inspect import getmembers, isfunction
            self._all_modules = sorted(name for name, _ in getmembers(default_modules, isfunction))
            self._enabled_modules = []

    def on_run_requested(self, message: RunRequested) -> None:
        self.push_screen(
            ProgressScreen(
                enabled_modules=message.enabled_modules,
                dry_run=message.dry_run,
                config_path=self._config_path,
            )
        )

    def on_cleanup_done(self, message: CleanupDone) -> None:
        self.push_screen(ResultsScreen(result=message.result))
```
